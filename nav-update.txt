// ── NAVIGATION ────────────────────────────────────────────────
function navigate(page){
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
  document.getElementById('page-'+page).classList.add('active');
  document.querySelectorAll('.nav-links a').forEach(a=>{
    a.classList.toggle('active',a.dataset.page===page);
  });
  window.scrollTo({top:0,behavior:'smooth'});
  triggerReveal();
  // Update URL with clean path
  if(page==='home'){
    history.pushState({page},'','/');
  } else {
    history.pushState({page},'','/'+page);
  }
  // Initialize blog filters if on blog page
  if(page==='blog') initBlogFilters();
}

// ── POPSTATE ROUTING ──────────────────────────────────────────────
window.addEventListener('popstate',(e)=>{
  const page=e.state?.page||'home';
  const inactive=document.querySelectorAll('.page.active');
  inactive.forEach(p=>p.classList.remove('active'));
  document.getElementById('page-'+page).classList.add('active');
  document.querySelectorAll('.nav-links a').forEach(a=>{
    a.classList.toggle('active',a.dataset.page===page);
  });
  triggerReveal();
  if(page==='blog') initBlogFilters();
});

// On page load, check current path and navigate accordingly
document.addEventListener('DOMContentLoaded',()=>{
  const path=window.location.pathname;
  let page='home';
  if(path==='/') page='home';
  else if(path.startsWith('/')) page=path.substring(1).split('/')[0]||'home';
  navigate(page);
});
